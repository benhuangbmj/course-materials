Which description could represent the graph of $f(x) = 4x^2(x+a) - x - a$, if $a$ is an integer?
\begin{enumerate}[label = (\arabic*), itemsep=2em]
\item 
As $x\to-\infty$, $f(x) \to \infty$, as $x\to\infty$, $f(x)\to\infty$, and the graph has $3$ $x$-intercepts.
\item
As $x\to-\infty$, $f(x) \to -\infty$, as $x\to\infty$, $f(x)\to\infty$, and the graph has $3$ $x$-intercepts.
\item
As $x\to-\infty$, $f(x) \to \infty$, as $x\to\infty$, $f(x)\to-\infty$, and the graph has $4$ $x$-intercepts.
\item
As $x\to-\infty$, $f(x) \to -\infty$, as $x\to\infty$, $f(x)\to\infty$, and the graph has $4$ $x$-intercepts.
\end{enumerate}

$p(x) = (x^2-9)(x-2)$
\item
$p(x) = x^3 - 2x^2 + 9 x + 18$
\item
$p(x) = (x^2 + 9)(x-2)$
\item
$p(x) = x^3 + 2x^2 - 9x - 18$
\end{enumerate}
\item
What is the solution set of the following system of equations?
\begin{align*}
y &= 3x+6\\
y &= (x+4)^2 - 10

\item
$\left\{(-5,-9)\right\}$
\item
$\left\{(5,21)\right\}$
\item
$\left\{(0,6),\ (-5,-9)\right\}$
\item
$\left\{(0,6),\ (5,21)\right\}$
\end{enumerate}
\item Which expression is equivalent to $(x+2)^2 - 5(x+2) + 6$?

$(x-3)(x-2)$
\item
$(x-4)(x+3)$
\item
$(x-6)(x+1)$
\end{enumerate}

function groupByUL(t)
local output = {}
local ind = {}
function makeKey(unit, lesson)
return "Unit " .. unit .. " Lesson " .. lesson
end
function checkForKey(key, t)
return  t[key] ~= nil
end
for i = #t, 1, -1 do
local curr = t[i]
local currKey = makeKey(curr.unit, curr.lesson)
if checkForKey(currKey, output) then
table.insert(output[currKey], curr)
else
output[currKey] = {curr}
table.insert(ind, currKey)
end
end
return output, ind
end
ftcsv = require("../ftcsv")
local learning_targets = ftcsv.parse("learning_targets.csv")
local grouped_LT, group_index = groupByUL(learning_targets)
for i=1, #group_index do
    local targets = grouped_LT[group_index[i]]
    tex.sprint("\\begin{frame}\\frametitle{\\bf "..group_index[i].." ".. targets[1].title.."}")
    tex.sprint("I can")
    tex.sprint("\\vspace{\\stretch{1}}")
    tex.sprint("\\begin{itemize}[leftmargin=*, label=$\\bullet$]")
    for _ , target in ipairs(targets) do	
    tex.sprint("\\item")
    tex.sprint(target.learning_target)
    tex.sprint("\\vspace{\\stretch{1}}")
    end
    tex.sprint("\\end{itemize}")
    tex.sprint("\\end{frame}")
end
